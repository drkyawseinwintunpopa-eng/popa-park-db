    /***************************************************************************
     * 7. PART 2 LOGIC: DETERMINISTIC SCHEDULE GENERATOR (2026-11-25 – 2031-03-31)
     ***************************************************************************/
    function generateAllTasks() {
      const tasks = [];
      const shiftOn = syncData.settings.shiftWeekends;

      function resolveDate(baseISO) {
        let effectiveISO = baseISO;
        let shifted = false;
        if (shiftOn) {
          const dow = getDow(baseISO);
          if (dow === 6) { effectiveISO = addDaysISO(baseISO, -1); shifted = true; }
          else if (dow === 0) { effectiveISO = addDaysISO(baseISO, -2); shifted = true; }
          if (effectiveISO < RANGE_START) effectiveISO = RANGE_START;
          if (effectiveISO > RANGE_END) effectiveISO = RANGE_END;
        }
        return { effectiveISO, shifted };
      }

      function pushBuiltInTask(baseTask) {
        const delRec = syncData.deletedTasks && syncData.deletedTasks[baseTask.id];
        if (delRec && delRec.deleted) return;

        const task = { ...baseTask, isCustom: false, isEdited: false };
        const ov = syncData.taskOverrides && syncData.taskOverrides[task.id];
        if (ov && !ov.reset) {
          if (ov.date && ov.date >= RANGE_START && ov.date <= RANGE_END) {
            if (ov.date !== task.date) task.shifted = false;
            task.date = ov.date;
          }
          if (ov.dept) task.dept = normalizeDept(ov.dept);
          if (ov.title) task.title = ov.title;
          if (ov.recipient) task.recipient = ov.recipient;
          if (ov.freq) task.freq = ov.freq;
          if (ov.priority) task.priority = ov.priority === 'high' ? 'high' : 'normal';
          task.isEdited = true;
        }
        tasks.push(task);
      }

      // A. Weekly Tasks (Strictly Tuesday / Thursday)
      let cursor = RANGE_START;
      while (cursor <= RANGE_END) {
        const dow = getDow(cursor);
        for (const r of WEEKLY_RULES) {
          if (r.dow === dow) {
            pushBuiltInTask({
              id: `task_${cursor}_${r.code}`,
              date: cursor,
              baseDate: cursor,
              shifted: false,
              dept: r.dept,
              title: r.title,
              recipient: r.recipient,
              freq: r.freq,
              priority: r.priority
            });
          }
        }
        cursor = addDaysISO(cursor, 1);
      }

      // B & C. Monthly, Quarterly, Semi-Annual & Annual Tasks across all 53 months
      for (const mk of MONTH_KEYS) {
        const [y, m] = mk.split('-').map(Number);
        const daysInMonth = new Date(y, m, 0, 12, 0, 0).getDate();
        const ctx = buildCtx(y, m);

        for (const r of MONTHLY_RULES) {
          const baseISO = toISO(y, m, Math.min(r.day, daysInMonth));
          if (baseISO < RANGE_START || baseISO > RANGE_END) continue;
          const { effectiveISO, shifted } = resolveDate(baseISO);
          pushBuiltInTask({
            id: `task_${baseISO}_${r.code}`,
            date: effectiveISO,
            baseDate: baseISO,
            shifted,
            dept: r.dept,
            title: interpolate(r.title, ctx),
            recipient: r.recipient,
            freq: r.freq,
            priority: r.priority
          });
        }

        for (const r of PERIODIC_RULES) {
          if (!r.months.includes(m)) continue;
          const baseISO = toISO(y, m, Math.min(r.day, daysInMonth));
          if (baseISO < RANGE_START || baseISO > RANGE_END) continue;
          const { effectiveISO, shifted } = resolveDate(baseISO);
          pushBuiltInTask({
            id: `task_${baseISO}_${r.code}`,
            date: effectiveISO,
            baseDate: baseISO,
            shifted,
            dept: r.dept,
            title: interpolate(r.title, ctx),
            recipient: r.recipient,
            freq: r.freq,
            priority: r.priority
          });
        }
      }

      // D. Active Custom Tasks
      for (const ct of syncData.customTasks) {
        if (ct.deleted) continue;
        if (ct.date >= RANGE_START && ct.date <= RANGE_END) {
          tasks.push({
            id: ct.id,
            date: ct.date,
            baseDate: ct.date,
            shifted: false,
            dept: normalizeDept(ct.dept),
            title: ct.title,
            recipient: ct.recipient,
            freq: ct.freq || 'ထပ်တိုးလုပ်ငန်း',
            priority: ct.priority === 'high' ? 'high' : 'normal',
            isCustom: true,
            isEdited: false
          });
        }
      }

      // Sort: Date ASC -> Priority High -> Dept Order -> ID
      tasks.sort((a, b) => {
        if (a.date !== b.date) return a.date.localeCompare(b.date);
        if (a.priority !== b.priority) return a.priority === 'high' ? -1 : 1;
        const da = DEPT_ORDER.indexOf(a.dept);
        const db = DEPT_ORDER.indexOf(b.dept);
        if (da !== db) return da - db;
        return a.id.localeCompare(b.id);
      });

      allTasksCache = tasks;
      return tasks;
    }

    function getTaskState(taskId) {
      return syncData.taskStates[taskId] || { done: false, completedAt: null, letterNo: '', note: '', updatedAt: 0 };
    }

    function getScopedTasks() {
      if (uiState.selectedMonth === 'ALL') return allTasksCache;
      if (uiState.selectedMonth.startsWith('FY:')) {
        const range = getFiscalYearRange(uiState.selectedMonth.slice(3));
        return allTasksCache.filter(t => t.date >= range.startISO && t.date <= range.endISO);
      }
      return allTasksCache.filter(t => t.date.startsWith(uiState.selectedMonth));
    }

    function filterTasks(list) {
      const q = uiState.searchQuery.trim().toLowerCase();
      const todayISO = getTodayISO();
      return list.filter(t => {
        const st = getTaskState(t.id);
        if (uiState.deptFilter !== 'ALL' && t.dept !== uiState.deptFilter) return false;
        if (uiState.statusFilter === 'PENDING' && st.done) return false;
        if (uiState.statusFilter === 'DONE' && !st.done) return false;
        if (uiState.statusFilter === 'OVERDUE' && (st.done || t.date >= todayISO)) return false;
        if (q) {
          const hay = `${t.title} ${t.recipient} ${t.dept} ${t.freq} ${st.letterNo} ${st.note} ${formatMMShortDate(t.date)}`.toLowerCase();
          if (!hay.includes(q)) return false;
        }
        return true;
      });
    }

    function handleImportJSON(e) {
      const file = e.target.files && e.target.files[0];
      if (!file) return;
      const reader = new FileReader();
      reader.onload = (ev) => {
        try {
          const parsed = JSON.parse(ev.target.result);
          // Support Encrypted .popaenc Backup Vault files
          if (parsed && parsed.format === 'POPA_ENC_V1') {
            pendingEncryptedFilePayload = parsed;
            openEngGateModal('decryptImport', 'ကုဒ်ဝှက် Backup ဖွင့်ခြင်း', null);
            return;
          }
          const payload = parsed.data ? parsed.data : parsed;
          if (!payload || typeof payload !== 'object') throw new Error('Invalid JSON');

          if (payload.taskStates || payload.taskOverrides || payload.deletedTasks || payload.dbConfigs || payload.customTasks) {
            mergeRemoteState(payload);
          } else {
            throw new Error('Unrecognized JSON');
          }

          saveLocalData();
          renderAll();
          scheduleCloudPush();
          logAudit('SYSTEM', 'JSON Backup ပြန်သွင်းပြီး', file.name, 'success');
          showToast('Backup ဒေတာများ အောင်မြင်စွာ ပေါင်းစပ်သွင်းပြီးပါပြီ', 'success');
        } catch (_) {
          showToast('Backup ဖိုင်ပုံစံ မမှန်ကန်ပါ (.JSON သို့ .popaenc ဖိုင်မှန်ရွေးပါ)', 'error');
        }
      };
      reader.readAsText(file, 'utf-8');
    }

    /** Live Analog & Digital Clock Engine (Top-Right Header Widget) */
    let liveClockTimer = null;

    function getMyanmarDayPeriod(hour24) {
      if (hour24 >= 5 && hour24 < 12) return 'နံနက်';
      if (hour24 >= 12 && hour24 < 16) return 'နေ့လယ်';
      if (hour24 >= 16 && hour24 < 19) return 'ညနေ';
      return 'ည';
    }

    function updateLiveClocks() {
      const now = new Date();
      const h24 = now.getHours();
      const m = now.getMinutes();
      const s = now.getSeconds();

      // Analog hand angles around SVG center (50, 50)
      const secAngle = s * 6;
      const minAngle = (m + s / 60) * 6;
      const hourAngle = ((h24 % 12) + m / 60 + s / 3600) * 30;

      const hrEl = document.getElementById('analogHourHand');
      const minEl = document.getElementById('analogMinuteHand');
      const secEl = document.getElementById('analogSecondHand');
      if (hrEl) hrEl.setAttribute('transform', `rotate(${hourAngle} 50 50)`);
      if (minEl) minEl.setAttribute('transform', `rotate(${minAngle} 50 50)`);
      if (secEl) secEl.setAttribute('transform', `rotate(${secAngle} 50 50)`);

      // Digital time display (12H AM/PM or 24H)
      const is24 = Boolean(uiState.clock24h);
      const dispH = is24 ? h24 : (h24 % 12 || 12);
      const ampm = h24 >= 12 ? 'PM' : 'AM';
      const period = getMyanmarDayPeriod(h24);

      const hmEl = document.getElementById('digitalClockHM');
      const secTxtEl = document.getElementById('digitalClockSec');
      const ampmEl = document.getElementById('digitalClockAmPm');
      const periodEl = document.getElementById('digitalClockPeriod');
      const modeTagEl = document.getElementById('digitalClockModeTag');
      const mmEl = document.getElementById('digitalClockMM');

      if (hmEl) hmEl.textContent = `${pad2(dispH)}:${pad2(m)}`;
      if (secTxtEl) secTxtEl.textContent = pad2(s);
      if (ampmEl) {
        ampmEl.textContent = ampm;
        ampmEl.classList.toggle('hidden', is24);
      }
      if (periodEl) periodEl.textContent = `${period}ပိုင်း`;
      if (modeTagEl) modeTagEl.textContent = is24 ? '24H' : '12H';
      if (mmEl) mmEl.textContent = `${period} ${toMM(pad2(dispH))}:${toMM(pad2(m))}:${toMM(pad2(s))}`;
    }

    function checkLiveDateTransition() {
      const nowISO = getTodayISO();
      if (nowISO !== lastCheckedDateISO) {
        lastCheckedDateISO = nowISO;
        renderKpisAndDeptProgress();
        renderActiveView();
        updateNotificationBadge();

      if (securityConfig.appLockEnabled && securityConfig.appPassHash && !isAppLocked) {
        const limitMin = Number(securityConfig.autoLockMin) || 0;
        if (limitMin > 0 && (Date.now() - lastActivityAt) >= limitMin * 60 * 1000) {
          lockAppNow();
        }
      }
      if (engUnlockedUntil > 0 && Date.now() >= engUnlockedUntil) {
        engUnlockedUntil = 0;
        updateSecurityIndicators();
      }
    }
